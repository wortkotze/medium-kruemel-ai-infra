Entwickler-Schnellstart

Befehle zur Steuerung, Überprüfung und Nutzung der lokalen Infrastruktur: 

from openai import OpenAI

client = OpenAI(
    base_url="http://localhost:4000/v1",
    api_key="<LITELLM_MASTER_KEY>"
)

response = client.chat.completions.create(
    model="local-general",   # Routing auf lokales Llama 3.1 via LiteLLM
    messages=[{"role": "user", "content": "Prüfe den Systemzustand."}],
    temperature=0.2
)
print(response.choices[0].message.content)
from langchain_openai import ChatOpenAI
from langfuse.langchain import CallbackHandler

langfuse_handler = CallbackHandler(
    host="http://localhost:3000",
    public_key="<LANGFUSE_PUBLIC_KEY>",
    secret_key="<LANGFUSE_SECRET_KEY>"
)

llm = ChatOpenAI(
    model="local-coder",
    openai_api_base="http://localhost:4000/v1",
    openai_api_key="<LITELLM_MASTER_KEY>",
    callbacks=[langfuse_handler]
)

response = llm.invoke("Erstelle ein Bash-Backup-Skript.")
curl -X POST http://localhost:4000/v1/chat/completions \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer <LITELLM_MASTER_KEY>" \
  -d '{
    "model": "local-coder",
    "messages": [{"role": "user", "content": "Hello World"}],
    "max_tokens": 64
  }'
